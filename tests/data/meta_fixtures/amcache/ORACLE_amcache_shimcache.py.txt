"""Independent reader (regipy): amcache plugin / ShimCacheParser. Usage: python -P ORACLE... <file.bin>
Prints JSON: {"amcache": [entries in regipy order]} | {"shimcache": {controlset: [{path, last_mod_date}]}} |
{"shim_blob": [...]} | {"error": ...}."""
import json
import sys

from regipy.registry import RegistryHive
from regipy.plugins.amcache.amcache import AmCachePlugin
from regipy.plugins.system.external.ShimCacheParser import get_shimcache_entries

p = sys.argv[1]
b = open(p, "rb").read()
if b[:4] != b"regf":
    try:
        print(json.dumps({"shim_blob": list(get_shimcache_entries(b, as_json=True) or [])}))
    except Exception as e:  # noqa: BLE001
        print(json.dumps({"error": "not regf / not shim: %r" % e}))
    sys.exit()
h = RegistryHive(p)
out = {}
try:
    pl = AmCachePlugin(h, as_json=True)
    pl.run()
    if pl.entries:
        out["amcache"] = pl.entries
except Exception as e:  # noqa: BLE001
    out["amcache_error"] = repr(e)
try:
    cur = h.get_key("\\Select").get_value("Current")
    out["select_current"] = cur
    for cs in ("ControlSet001", "ControlSet002"):
        v = h.get_key("\\%s\\Control\\Session Manager\\AppCompatCache" % cs).get_value("AppCompatCache")
        out.setdefault("shimcache", {})[cs] = list(get_shimcache_entries(v, as_json=True))
except Exception as e:  # noqa: BLE001
    out["shimcache_error"] = repr(e)
print(json.dumps(out, indent=1, default=str))
