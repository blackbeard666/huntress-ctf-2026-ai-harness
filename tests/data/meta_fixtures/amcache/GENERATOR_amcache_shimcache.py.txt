"""Stdlib struct writer for Amcache.hve / SYSTEM hives (nk/lh/lf/ri/vk/value-list cells + `db` big data) and raw
Win10 AppCompatCache blobs. Usage: python3 -I GENERATOR_amcache_shimcache.py.txt <outdir>
Writes: amcache_win10.bin (Root\\InventoryApplicationFile x3 + InventoryApplication decoy), amcache_single.bin (one
entry, legacy REG_SZ hex Size), system_shim.bin (SYSTEM: Select\\Current=2, ControlSet001 decoy cache, ControlSet002
cache with 0x34 header > 16344 bytes -> db), shim_blob_win10.bin (raw value, 0x30 header), not_shim.bin (PNG bytes)."""

class Hive:
    def __init__(self):
        self.cells = bytearray()

    def alloc(self, payload):
        off = 0x20 + len(self.cells)
        size = (len(payload) + 4 + 7) & ~7
        self.cells += struct.pack("<i", -size) + payload + b"\0" * (size - 4 - len(payload))
        return off

    def patch(self, off, at, fmt, *vals):
        struct.pack_into(fmt, self.cells, off - 0x20 + 4 + at, *vals)


def utf16(s):
    return s.encode("utf-16-le")


def lhash(n):
    v = 0
    for c in n.upper():
        v = (v * 37 + ord(c)) & 0xFFFFFFFF
    return v


def build(seq, base_ts, root, fname):
    """root = (name, ts, [values], [children]); value = (name, type, data bytes). lh lists, children name-sorted."""
    h = Hive()
    sd = struct.pack("<BBHIIII", 1, 0, 0x8004, 0, 0, 0, 0)
    sk = 0x20 + ((4 + 0x4C + len(root[0]) + 7) & ~7)   # the root nk is the FIRST cell (regipy/Windows), sk right after

    def make_sk():
        assert h.alloc(b"sk\0\0" + struct.pack("<IIII", 0, 0, 1, len(sd)) + sd) == sk
        h.patch(sk, 4, "<II", sk, sk)

    def key(node, parent):
        name, ts, values, children = node
        nm = name.encode("latin-1")
        flags = 0x20 | (0x2C if parent is None else 0)
        nk = h.alloc(b"nk" + struct.pack("<HQIIIIIIIIIIIIIIIHH", flags, ts, 0, parent or 0, 0, 0, 0xFFFFFFFF,
                                         0xFFFFFFFF, 0, 0xFFFFFFFF, sk, 0xFFFFFFFF, 0, 0, 0, 0, 0, len(nm), 0) + nm)
        if parent is None:
            make_sk()
        children = sorted(children, key=lambda c: c[0].upper())
        offs = [key(c, nk) for c in children]
        if offs:
            lst = h.alloc(b"lh" + struct.pack("<H", len(offs)) + b"".join(
                struct.pack("<II", o, lhash(c[0])) for o, c in zip(offs, children)))
            h.patch(nk, 0x14, "<I", len(offs))
            h.patch(nk, 0x1C, "<I", lst)
        if values:
            vks = []
            for vn, vt, vd in values:
                vnb = vn.encode("latin-1")
                if len(vd) <= 4:
                    size, doff = 0x80000000 | len(vd), struct.unpack("<I", vd.ljust(4, b"\0"))[0]
                elif len(vd) > 16344:        # big data: db -> segment list -> 16344-byte segments
                    segs = [h.alloc(vd[i:i + 16344]) for i in range(0, len(vd), 16344)]
                    sl = h.alloc(b"".join(struct.pack("<I", s) for s in segs))
                    size, doff = len(vd), h.alloc(b"db" + struct.pack("<HII", len(segs), sl, 0))
                else:
                    size, doff = len(vd), h.alloc(vd)
                vks.append(h.alloc(b"vk" + struct.pack("<HIIIHH", len(vnb), size, doff, vt, 1, 0) + vnb))
            vl = h.alloc(b"".join(struct.pack("<I", o) for o in vks))
            h.patch(nk, 0x24, "<II", len(vks), vl)
        return nk

    root_off = key(root, None)
    body = bytes(h.cells)
    hbin_size = (0x20 + len(body) + 8 + 0xFFF) & ~0xFFF
    free = hbin_size - 0x20 - len(body)
    hbin = b"hbin" + struct.pack("<IIQQI", 0, hbin_size, 0, base_ts, 0) + body + struct.pack("<i", free) + b"\0" * (free - 4)
    fn = utf16(fname)[:64].ljust(64, b"\0")
    base = b"regf" + struct.pack("<IIQIIIIIII", seq, seq, base_ts, 1, 5, 0, 1, root_off, hbin_size, 1) + fn
    base = base.ljust(508, b"\0")
    ck = 0
    for i in range(0, 508, 4):
        ck ^= struct.unpack_from("<I", base, i)[0]
    return base + struct.pack("<I", ck) + b"\0" * (0x1000 - 512) + hbin


def sz(s):
    return utf16(s + "\0")


def q(v):
    return struct.pack("<Q", v)


def ft(unix):
    return (unix + 11644473600) * 10000000

def shim(header, entries, pad=0):
    """Win10 AppCompatCache blob: header (size u32 @0 + stats), then 10ts entries."""
    blob = struct.pack("<I", header) + b"\0" * (header - 4)
    for path, ts, data in entries:
        p = utf16(path)
        body = struct.pack("<H", len(p)) + p + q(ts) + struct.pack("<I", len(data)) + data
        blob += b"10ts" + struct.pack("<II", zlib.crc32(body), len(body)) + body
    return blob
