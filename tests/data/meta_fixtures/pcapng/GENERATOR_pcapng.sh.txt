# pk.txt = 3 Ethernet ARP frames as a text2pcap hex dump, each preceded by "2026-03-14 09:26:53.589793" style stamps

TZ=UTC text2pcap -q -F pcapng -t "%Y-%m-%d %H:%M:%S.%f" -N huntress0 pk.txt eth_ng.pcapng
TZ=UTC text2pcap -q -F pcapng -t "%Y-%m-%d %H:%M:%S.%f" -N wlan_mon -l 105 pk.txt ifb.pcapng
editcap -F nsecpcap eth_us.pcap eth_ns.pcap
editcap -T rawip4 -C 14 -F pcap -s 96 eth_us.pcap raw_cut.pcap
editcap -F pcapng eth_us.pcap conv.pcapng
editcap -F snoop eth_us.pcap decoy_snoop.pcap          # Sun snoop impostor named .pcap
mergecap -F pcapng -w multi.pcapng eth_ng.pcapng ifb.pcapng
