def tarbytes(entries):
    bio = io.BytesIO()
    with tarfile.open(fileobj=bio, mode="w", format=tarfile.USTAR_FORMAT) as tf:

            tf.addfile(ti, io.BytesIO(data))
    return bio.getvalue()
