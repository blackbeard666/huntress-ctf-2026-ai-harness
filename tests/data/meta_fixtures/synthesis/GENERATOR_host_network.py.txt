"""Generator for the synthesis_network fixtures (host claims vs capture traffic).
usage: .venv/bin/python -I GENERATOR_host_network.py.txt <outdir>
Captures are written with the dpkt pcap / pcapng WRITERS (packets built with dpkt.ethernet/ip/tcp/udp/dns);
host logs are plain text written here (netstat -ano table, W3C pfirewall.log, Zeek conn.log TSV); the Sysmon EVTX member
is the existing fixture tests/data/meta_fixtures/sysmon/hostnet_sysmon__remote_ips.bin (stdlib BinXML writer,
GENERATOR_host_net_sysmon.py.txt). Containers: stdlib zipfile."""
import io
import os
import socket
import sys
import zipfile

import dpkt

OUT = sys.argv[1]
HERE = os.path.dirname(os.path.abspath(__file__))
SYSMON = os.path.join(HERE, "..", "sysmon", "hostnet_sysmon__remote_ips.bin")


def eth(src, dst, l4, proto):
    ip = dpkt.ip.IP(src=socket.inet_aton(src), dst=socket.inet_aton(dst), p=proto, ttl=64)
    ip.data = l4
    ip.len = len(ip)
    return bytes(dpkt.ethernet.Ethernet(src=b"\x02" * 6, dst=b"\x04" * 6, type=0x0800, data=ip))


def tcp(src, sp, dst, dp, flags):
    return eth(src, dst, dpkt.tcp.TCP(sport=sp, dport=dp, flags=flags, seq=1000, off=5), 6)


def dnsq(src, sp, dst, name):
    q = dpkt.dns.DNS(id=0x1234, qd=[dpkt.dns.DNS.Q(name=name, type=1, cls=1)])
    return eth(src, dst, dpkt.udp.UDP(sport=sp, dport=53, data=bytes(q), ulen=8 + len(bytes(q))), 17)


def pcap(pkts, t0=1760000000):
    bio = io.BytesIO()
    w = dpkt.pcap.Writer(bio, snaplen=65535, linktype=1)
    for i, p in enumerate(pkts):
        w.writepkt(p, ts=t0 + i)
    return bio.getvalue()


def pcapng(pkts, t0=1760000000):
    bio = io.BytesIO()
    w = dpkt.pcapng.Writer(bio, snaplen=65535, linktype=1)
    for i, p in enumerate(pkts):
        w.writepkt(p, ts=t0 + i)
    return bio.getvalue()


SYN, SYNACK, ACK = 0x02, 0x12, 0x10
