# stdlib xml.etree WRITER with a PREFIXED namespace (t:Task) + ComHandler-only action, UTF-16 (BOM, LE host).
# Usage: python3 -I this.py OUTFILE
import sys, xml.etree.ElementTree as ET
NS = "http://schemas.microsoft.com/windows/2004/02/mit/task"
ET.register_namespace("t", NS)
q = lambda n: "{%s}%s" % (NS, n)
task = ET.Element(q("Task"), {"version": "1.3"})
reg = ET.SubElement(task, q("RegistrationInfo"))
ET.SubElement(reg, q("Source")).text = "$(@%SystemRoot%\\system32\\wininet.dll,-16771)"
ET.SubElement(reg, q("Author")).text = "$(@%SystemRoot%\\system32\\wininet.dll,-16772)"
ET.SubElement(reg, q("URI")).text = "\\Microsoft\\Windows\\Wininet\\CacheTask"
tr = ET.SubElement(task, q("Triggers"))
ET.SubElement(ET.SubElement(tr, q("LogonTrigger")), q("Enabled")).text = "true"
pr = ET.SubElement(ET.SubElement(task, q("Principals")), q("Principal"), {"id": "AnyUser"})
ET.SubElement(pr, q("GroupId")).text = "S-1-5-32-545"
ET.SubElement(ET.SubElement(task, q("Settings")), q("Hidden")).text = "false"
ch = ET.SubElement(ET.SubElement(task, q("Actions"), {"Context": "AnyUser"}), q("ComHandler"))
ET.SubElement(ch, q("ClassId")).text = "{0358B920-0AC7-461F-98F4-58E32CD89148}"
ET.ElementTree(task).write(sys.argv[1], encoding="utf-16", xml_declaration=True)
