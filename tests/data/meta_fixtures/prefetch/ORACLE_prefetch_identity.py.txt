"""Independent oracle for the Prefetch identity fixtures (does NOT import lib.meta). libyal pyscca (tools/pylib-scca):
header name/hash, version, filename strings, every volume (device path, serial, creation time); windowsprefetch 4.0.3
(tools/pylib-windowsprefetch, PoorBillionaire): directory strings per volume, fed the SCCA bytes (MAM payloads
decompressed with dissect.util.compression.lzxpress_huffman first: its own MAM path is Windows-ctypes only;
a v30 SCCA is run through that MAM branch's parse sequence); the path
hash = libscca "Prefetch hash" Python transcribed verbatim (XP for v17, Vista for v23, 2008 for v26+), device volume
number for \\VOLUME{} paths = the HARDDISKVOLUME<n> whose hash equals pyscca's prefetch_hash.
Usage: .venv/bin/python -I ORACLE_prefetch_identity.py.txt <file.bin> <member name> -> one JSON object."""
import datetime
import json
import os
import re
import struct
import sys
import tempfile

sys.path.insert(0, os.path.join(os.getcwd(), "tools", "pylib-scca"))
sys.path.insert(0, os.path.join(os.getcwd(), "tools", "pylib-windowsprefetch"))
import pyscca  # noqa: E402
from dissect.util.compression import lzxpress_huffman  # noqa: E402
from windowsprefetch.windowsprefetch import Prefetch  # noqa: E402


def scca_xp_hash_function(filename):
    hash_value = 0
    for character in filename:
        hash_value = ((hash_value * 37) + character) % 0x100000000
    hash_value = (hash_value * 314159269) % 0x100000000
    if hash_value > 0x80000000:
        hash_value = 0x100000000 - hash_value
    return (abs(hash_value) % 1000000007) % 0x100000000


def scca_vista_hash_function(filename):
    hash_value = 314159
    for character in filename:
        hash_value = ((hash_value * 37) + character) % 0x100000000
    return hash_value


def scca_2008_hash_function(filename):
    hash_value = 314159
    filename_index = 0
    filename_length = len(filename)
    while filename_index + 8 < filename_length:
        character_value = filename[filename_index + 1] * 37
        character_value += filename[filename_index + 2]
        character_value *= 37
        character_value += filename[filename_index + 3]
        character_value *= 37
        character_value += filename[filename_index + 4]
        character_value *= 37
        character_value += filename[filename_index + 5]
        character_value *= 37
        character_value += filename[filename_index + 6]
        character_value *= 37
        character_value += filename[filename_index] * 442596621
        character_value += filename[filename_index + 7]
        hash_value = ((character_value - (hash_value * 803794207)) % 0x100000000)
        filename_index += 8
    while filename_index < filename_length:
        hash_value = (((37 * hash_value) + filename[filename_index]) % 0x100000000)
        filename_index += 1
    return hash_value


path, member = sys.argv[1], sys.argv[2]
d = open(path, "rb").read()
out = {}
m = re.match(r"^(.+)-([0-9A-Fa-f]{8})\.pf$", member, re.I)
out["filename_hash_from_name"] = "0x" + m.group(2).lower() if m else "none"
try:
    f = pyscca.open(path)
except IOError:
    print(json.dumps(dict.fromkeys(["executable_path", "filename_hash_from_name", "volume_serials"], "none")))
    sys.exit(0)
ver, name, hdr = f.format_version, f.executable_filename, f.prefetch_hash
stem = m.group(1) if m else re.sub(r"\.pf$", "", member, flags=re.I)
out["filename_stem_matches_header"] = "true" if stem.upper() == name.upper() else "false"
fns = [f.get_filename(i) for i in range(f.number_of_filenames)]
out["filename_strings_count"] = str(len(fns))
exe = [p for p in fns if p.split("\\")[-1].upper() == name.upper()]
out["executable_path"] = exe[0] if exe else "none"
out["executable_directory"] = exe[0].rsplit("\\", 1)[0] if exe else "none"
vols = [f.get_volume_information(i) for i in range(f.number_of_volumes)]
out["volume_serials"] = json.dumps(["0x%08x" % v.serial_number for v in vols], separators=(",", ":"))
out["volume_device_paths"] = json.dumps([v.device_path for v in vols], separators=(",", ":"), ensure_ascii=False)
out["volume_creation_times_utc"] = json.dumps([(datetime.datetime(1601, 1, 1) + datetime.timedelta(
    microseconds=v.get_creation_time_as_integer() // 10)).strftime("%Y-%m-%dT%H:%M:%SZ") for v in vols],
    separators=(",", ":"))
f.close()
scca = lzxpress_huffman.decompress(d[8:])[:struct.unpack_from("<I", d, 4)[0]] if d[:4] == b"MAM\x04" else d
with tempfile.NamedTemporaryFile(suffix=".pf", delete=False) as t:
    t.write(scca)
try:
    if struct.unpack_from("<I", scca, 0)[0] in (30, 31):   # windowsprefetch reaches v30 only via its MAM branch:
        wp = Prefetch.__new__(Prefetch)                       # run that branch's own parse sequence on the SCCA
        with open(t.name, "rb") as fh:
            wp.parseHeader(fh)
            wp.fileInformation26(fh)
            wp.metricsArray23(fh)
            wp.traceChainsArray30(fh)
            wp.volumeInformation30(fh)
    else:
        wp = Prefetch(t.name)
    dirs = sorted(s.rstrip("\x00") for vol in wp.directoryStringsArray for s in vol)
finally:
    os.unlink(t.name)
out["directory_strings"] = json.dumps(dirs, separators=(",", ":"))
out["directory_count"] = str(len(dirs))
if exe:
    hf = scca_xp_hash_function if ver == 17 else scca_vista_hash_function if ver == 23 else scca_2008_hash_function
    p = exe[0]
    cands = [p] if not p.upper().startswith("\\VOLUME{") else \
        ["\\DEVICE\\HARDDISKVOLUME%d%s" % (n, p[p.index("}") + 1:]) for n in range(65)]
    hits = [(q, hf(q.upper().encode("utf-16-le"))) for q in cands]
    hit = [x for x in hits if x[1] == hdr] or (hits if len(hits) == 1 else [])
    out["executable_device_path"] = hit[0][0] if hit else "None"
    out["computed_path_hash_u32_hex"] = "0x%08x" % hit[0][1] if hit else "None"
    out["path_hash_matches"] = "true" if hit and hit[0][1] == hdr else "false"
print(json.dumps(out, ensure_ascii=False))
