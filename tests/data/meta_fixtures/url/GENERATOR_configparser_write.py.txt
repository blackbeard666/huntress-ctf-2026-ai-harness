# Writes .url fixtures with the stdlib configparser WRITER (CRLF like Windows). usage: python -I THIS OUTDIR
import configparser, io, sys
out = sys.argv[1]
def write(name, sections, defaults=None):
    cp = configparser.RawConfigParser(defaults=defaults)
    cp.optionxform = str
    for sec, kv in sections:
        cp.add_section(sec)
        for k, v in kv:
            cp.set(sec, k, v)
    buf = io.StringIO()
    cp.write(buf, space_around_delimiters=False)
    with open(f"{out}/{name}", "wb") as f:
        f.write(buf.getvalue().replace("\n", "\r\n").encode("cp1252"))
