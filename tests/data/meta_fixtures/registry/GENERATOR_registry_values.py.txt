"""Stdlib struct writer for registry hives carrying HOST-IDENTITY values (SYSTEM / SOFTWARE / SAM layouts).
Usage: python3 -I GENERATOR_registry_values.py.txt <outdir>
Cell/hbin/base-block writer copied verbatim from ../regf/GENERATOR_regf_keys.py.txt (build()); trees below:
  system_cs2.bin   : Select\\Current=2; ControlSet001 = DECOY host OLDHOST-01 / UTC, ControlSet002 = WS-ANALYST7 /
                     Pacific Standard Time (the answer must follow Select\\Current), root values Marker/Ordinal.
  software_cv.bin  : Classes + Microsoft\\Windows NT\\CurrentVersion (ProductName, InstallDate DWORD unix,
                     RegisteredOwner, CurrentBuild, MULTI_SZ + BINARY values).
  sam_account.bin  : SAM\\Domains\\Account V (binary, SID sub-authorities in the last 12 bytes).
Oracle: python-registry 1.3.1 (tools/pylib/Registry), an independent reader."""
import codecs
import struct
import sys

out = sys.argv[1]


class Hive:
    def __init__(self):
        self.cells = bytearray()

    def alloc(self, payload):
        off = 0x20 + len(self.cells)
        size = (len(payload) + 4 + 7) & ~7
        self.cells += struct.pack("<i", -size) + payload + b"\0" * (size - 4 - len(payload))
        return off

    def patch(self, off, at, fmt, *vals):
        struct.pack_into(fmt, self.cells, off - 0x20 + 4 + at, *vals)


def utf16(s):
    return s.encode("utf-16-le")


def build(seq, base_ts, root):
    """root = (name, ts, [values], [children], index_kind); value = (name, type, data bytes)."""
    h = Hive()
    sd = struct.pack("<BBHIIII", 1, 0, 0x8004, 0, 0, 0, 0)
    sk = h.alloc(b"sk\0\0" + struct.pack("<IIII", 0, 0, 1, len(sd)) + sd)
    h.patch(sk, 4, "<II", sk, sk)

    def key(node, parent):
        name, ts, values, children, kind = node
        comp = all(ord(c) < 128 for c in name) and not name.startswith("~")
        nm = name.encode("latin-1") if comp else utf16(name.lstrip("~"))
        flags = (0x20 if comp else 0) | (0x2C if parent is None else 0)
        nk = h.alloc(b"nk" + struct.pack("<HQIIIIIIIIIIIIIIIHH", flags, ts, 0, parent or 0, 0, 0, 0xFFFFFFFF,
                                         0xFFFFFFFF, 0, 0xFFFFFFFF, sk, 0xFFFFFFFF, 0, 0, 0, 0, 0, len(nm), 0) + nm)
        offs = [key(c, nk) for c in children]
        if offs:
            names = [c[0].lstrip("~") for c in children]
            if kind == "ri":
                half = (len(offs) + 1) // 2
                lists = [h.alloc(b"li" + struct.pack("<H", len(part)) + b"".join(struct.pack("<I", o) for o in part))
                         for part in (offs[:half], offs[half:]) if part]
                lst = h.alloc(b"ri" + struct.pack("<H", len(lists)) + b"".join(struct.pack("<I", o) for o in lists))
            elif kind == "lf":
                lst = h.alloc(b"lf" + struct.pack("<H", len(offs)) + b"".join(
                    struct.pack("<I", o) + n.encode("latin-1", "replace")[:4].ljust(4, b"\0") for o, n in zip(offs, names)))
            else:
                def lhash(n):
                    v = 0
                    for c in n.upper():
                        v = (v * 37 + ord(c)) & 0xFFFFFFFF
                    return v
                lst = h.alloc(b"lh" + struct.pack("<H", len(offs)) + b"".join(
                    struct.pack("<II", o, lhash(n)) for o, n in zip(offs, names)))
            h.patch(nk, 0x14, "<I", len(offs))
            h.patch(nk, 0x1C, "<I", lst)
        if values:
            vks = []
            for vn, vt, vd in values:
                vnb = vn.encode("latin-1")
                if len(vd) <= 4:
                    size, doff = 0x80000000 | len(vd), struct.unpack("<I", vd.ljust(4, b"\0"))[0]
                else:
                    size, doff = len(vd), h.alloc(vd)
                vks.append(h.alloc(b"vk" + struct.pack("<HIIIHH", len(vnb), size, doff, vt, 1, 0) + vnb))
            vl = h.alloc(b"".join(struct.pack("<I", o) for o in vks))
            h.patch(nk, 0x24, "<II", len(vks), vl)
        return nk

    root_off = key(root, None)
    body = bytes(h.cells)
    hbin_size = (0x20 + len(body) + 8 + 0xFFF) & ~0xFFF
    free = hbin_size - 0x20 - len(body)
    hbin = b"hbin" + struct.pack("<IIQQI", 0, hbin_size, 0, base_ts, 0) + body + struct.pack("<i", free) + b"\0" * (free - 4)
    fn = utf16("\\??\\C:\\Users\\ana\\ntuser.dat")[:64].ljust(64, b"\0")
    base = b"regf" + struct.pack("<IIQIIIIIII", seq, seq, base_ts, 1, 5, 0, 1, root_off, hbin_size, 1) + fn
    base = base.ljust(508, b"\0")
    ck = 0
    for i in range(0, 508, 4):
        ck ^= struct.unpack_from("<I", base, i)[0]
    base += struct.pack("<I", ck)
    return base.ljust(0x1000, b"\0") + hbin


def sz(s):
    return utf16(s + "\0")


def ft(unix):
    return (unix + 11644473600) * 10000000


def dw(n):
    return struct.pack("<I", n)


T = ft(1790000000)                        # 2026-09-21T14:13:20Z
cn = lambda host: ("ComputerName", T, [], [("ComputerName", T, [("ComputerName", 1, sz(host))], [], "lh"),  # noqa: E731
                                          ("ActiveComputerName", T, [("ComputerName", 1, sz(host))], [], "lh")], "lh")
tz = lambda name: ("TimeZoneInformation", T, [("Bias", 4, dw(480)), ("TimeZoneKeyName", 1, sz(name)),  # noqa: E731
                                              ("StandardName", 1, sz("@tzres.dll,-212"))], [], "lh")
cs1 = ("ControlSet001", T, [], [("Control", T, [], [cn("OLDHOST-01"), tz("UTC")], "lh")], "lh")
cs2 = ("ControlSet002", T, [], [("Control", T, [], [cn("WS-ANALYST7"), tz("Pacific Standard Time")], "lh")], "lh")
sel = ("Select", T, [("Current", 4, dw(2)), ("Default", 4, dw(2)), ("LastKnownGood", 4, dw(1))], [], "lh")
system = ("ROOT", T, [("Marker", 1, sz("HUNTRESS-HIVE-7781")), ("Ordinal", 4, dw(7781))],
          [cs1, cs2, ("MountedDevices", T, [], [], "lh"), sel], "lh")
open(out + "/system_cs2.bin", "wb").write(build(5, T, system))

cv = ("CurrentVersion", T, [("ProductName", 1, sz("Windows 10 Pro")), ("InstallDate", 4, dw(1700000000)),
                            ("RegisteredOwner", 1, sz("ana.lopez")), ("CurrentBuild", 1, sz("19045")),
                            ("Tags", 7, sz("alpha") + sz("beta") + b"\0\0"), ("DigitalProductId", 3, bytes(range(20)))],
      [], "lh")
software = ("CMI-CreateHive{199DAFC2-6F16-4946-BF90-5A3FC3A60902}", T, [],
            [("Classes", T, [], [], "lh"),
             ("Microsoft", T, [], [("Windows NT", T, [], [cv], "lh"), ("Windows", T, [], [], "lh")], "lh")], "lh")
open(out + "/software_cv.bin", "wb").write(build(6, T, software))

v = bytes(0xCC) + b"\x01\x04\x00\x00\x00\x00\x00\x05" + struct.pack("<IIII", 21, 3141592653, 2718281828, 1618033988)
sam = ("CsiTool-CreateHive-{00000000-0000-0000-0000-000000000000}", T, [],
       [("SAM", T, [("C", 3, b"\x07\x00\x01\x00")],
         [("Domains", T, [], [("Account", T, [("F", 3, bytes(0x50)), ("V", 3, v)],
                                [("Users", T, [], [], "lh")], "lh"),
                               ("Builtin", T, [("V", 3, bytes(64))], [], "lh")], "lh"),
          ("RXACT", T, [], [], "lh")], "lh")], "lh")
open(out + "/sam_account.bin", "wb").write(build(7, T, sam))
