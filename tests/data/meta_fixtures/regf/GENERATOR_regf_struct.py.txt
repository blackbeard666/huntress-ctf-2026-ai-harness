"""Stdlib struct writer for minimal Windows registry hives (regf base block + one hbin with a root nk cell + sk cell).
Usage: python3 -I GENERATOR_regf_struct.py.txt <outdir>"""
import struct
import sys

out = sys.argv[1]


def hive(seq1, seq2, ts, major, minor, fname, root_name, nk_ts, comp_name=True):
    name = root_name.encode("latin-1") if comp_name else root_name.encode("utf-16-le")
    # sk cell at hbin offset 0x20 (self-referencing list, minimal SD)
    sd = struct.pack("<BBHIIII", 1, 0, 0x8004, 0, 0, 0, 0)
    sk = b"sk" + b"\0\0" + struct.pack("<IIII", 0x20, 0x20, 1, len(sd)) + sd
    sk_cell = struct.pack("<i", -((len(sk) + 4 + 7) & ~7)) + sk
    sk_cell += b"\0" * ((-len(sk_cell)) % 8)
    nk_off = 0x20 + len(sk_cell)
    flags = 0x2C if comp_name else 0x0C
    nk = b"nk" + struct.pack("<HQIIIIIIIIIIIIIIIHH", flags, nk_ts, 0, 0xFFFFFFFF, 0, 0, 0xFFFFFFFF, 0xFFFFFFFF,
                             0, 0xFFFFFFFF, 0x20, 0xFFFFFFFF, 0, 0, 0, 0, 0, len(name), 0) + name
    nk_cell = struct.pack("<i", -((len(nk) + 4 + 7) & ~7)) + nk
    nk_cell += b"\0" * ((-len(nk_cell)) % 8)
    body = sk_cell + nk_cell
    free = 0x1000 - 0x20 - len(body)
    body += struct.pack("<i", free) + b"\0" * (free - 4)
    hbin = b"hbin" + struct.pack("<IIQQI", 0, 0x1000, 0, ts, 0) + body
    assert len(hbin) == 0x1000, len(hbin)
    fn = fname.encode("utf-16-le")[:64].ljust(64, b"\0")
    base = b"regf" + struct.pack("<IIQIIIIIII", seq1, seq2, ts, major, minor, 0, 1, nk_off, 0x1000, 1) + fn
    base = base.ljust(508, b"\0")
    ck = 0
    for i in range(0, 508, 4):
        ck ^= struct.unpack_from("<I", base, i)[0]
    base += struct.pack("<I", ck)
    base = base.ljust(0x1000, b"\0")
    return base + hbin


# FILETIMEs: 133550550450000000 = 2024-03-16T09:30:45Z ; ft2 = 2026-09-21T06:03:55.1234567Z
ft1 = 133550550450000000
ft2 = ((1789970635 + 11644473600) * 10000000) + 1234567
open(out + "/system_v13.bin", "wb").write(hive(5, 5, ft1, 1, 3, "emRoot\\System32\\Config\\SYSTEM", "ROOT", ft1 - 600 * 10**7))
open(out + "/ntuser_dirty_v15.bin", "wb").write(hive(1234, 1233, ft2, 1, 5, "\\??\\C:\\Users\\ana\\ntuser.dat",
                                                      "CsiTool-CreateHive-{00000000-0000-0000-0000-000000000000}", ft2, comp_name=False))
creg = b"CREG" + struct.pack("<IIIHHI", 0x10000, 0x20, 0, 1, 0, 0) + b"\0" * 4072
open(out + "/decoy_creg_win9x.bin", "wb").write(creg + b"RGKN" + b"\0" * 0x1C)
