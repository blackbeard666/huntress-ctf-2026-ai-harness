"""Independent oracle: python-registry 1.3.1 RegistryParse.REGFBlock (sequences, checksum, file type, hbin walk) +
regipy 6.x structs.REGF_HEADER / TRANSACTION_LOG (HvLE) and recovery._parse_dirt_block (DIRT). Prints JSON.
Usage: PYTHONPATH=tools/pylib .venv/bin/python -P ORACLE_regf_integrity.py.txt <file>"""
import io, json, os, sys, tempfile
from Registry import RegistryParse
from regipy.structs import REGF_HEADER, TRANSACTION_LOG
from regipy.recovery import _parse_dirt_block

p = sys.argv[1]
buf = open(p, "rb").read()
out = {}
try:
    r = RegistryParse.REGFBlock(buf.ljust(0x1000, b"\0"), 0, False)
except Exception as e:
    print(json.dumps({"error": "not regf: %r" % e}))
    sys.exit()
h = REGF_HEADER.parse(buf[:4096].ljust(4096, b"\0"))
out.update(primary=r.hive_sequence1(), secondary=r.hive_sequence2(), stored_checksum="0x%08x" % r.checksum(),
           checksum_valid=r.validate_checksum(), hbins_size=r.hbins_size(), file_type=r.file_type().value,
           regipy_file_type=h.file_type, file_name=h.file_name, file_size=os.path.getsize(p))
if r.file_type().value == 0:
    sizes = [hb._reloffset_next_hbin for hb in r.hbins()]
    out.update(hbin_count=len(sizes), hbin_sizes=sizes, hbin_walk=sum(sizes))
else:
    s = io.BytesIO(buf)
    s.seek(512)
    sig = buf[512:516]
    if sig == b"HvLE":
        seqs, dirty, off = [], 0, 512
        while buf[off:off + 4] == b"HvLE":
            e = TRANSACTION_LOG.parse(buf[off:])
            seqs.append(e.sequence_number)
            dirty += e.dirty_pages_count
            off += e.log_size
        out.update(log="HvLE", seqs=seqs, dirty_pages=dirty)
    elif sig == b"DIRT":
        with tempfile.NamedTemporaryFile() as t:
            t.write(b"\0" * 0x8000); t.flush()
            _, n = _parse_dirt_block(t.name, s, h.hive_bins_data_size)
        out.update(log="DIRT", dirty_pages=n)
print(json.dumps(out))
