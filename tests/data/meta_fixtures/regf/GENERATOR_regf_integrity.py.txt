"""Stdlib struct writer for registry hive INTEGRITY fixtures: dirty/truncated primary hive, bad-checksum primary,
new-format transaction log (HvLE entries), old-format log (DIRT vector), carved-hbin decoy.
Usage: python3 -I GENERATOR_regf_integrity.py.txt <outdir>"""
import struct
import sys

out = sys.argv[1]
FT = 133550550450000000          # 2024-03-16T09:30:45Z
FT2 = 134032421951234567         # 2025-09-21T06:03:15.1234567Z


def xsum(b):
    x = 0
    for i in range(0, 508, 4):
        x ^= struct.unpack_from("<I", b, i)[0]
    return 1 if x == 0 else 0xFFFFFFFE if x == 0xFFFFFFFF else x


def base(s1, s2, ts, minor, ftype, size, fname, full=True, corrupt=0):
    fn = fname.encode("utf-16-le")[:64].ljust(64, b"\0")
    b = b"regf" + struct.pack("<IIQIIIIIII", s1, s2, ts, 1, minor, ftype, 1, 0x20, size, 1) + fn
    b = b.ljust(508, b"\0")
    b += struct.pack("<I", xsum(b) ^ corrupt)
    return b.ljust(0x1000 if full else 512, b"\0")


def hbin(rel, size, ts):
    h = b"hbin" + struct.pack("<IIQQI", rel, size, 0, ts, 0)
    free = size - 0x20
    return h + struct.pack("<i", free) + b"\0" * (free - 4)


def hbins(sizes, ts):
    o, rel = b"", 0
    for s in sizes:
        o += hbin(rel, s, ts)
        rel += s
    return o


def w(name, b):
    with open("%s/%s" % (out, name), "wb") as fh:
        fh.write(b)


# A: dirty (673 vs 672), 3 x 4 KiB hbins declared 0x3000, file cut 0xC00 into the third hbin
a = base(673, 672, FT2, 6, 0, 0x3000, "\\??\\C:\\Users\\alice\\ntuser.dat") + hbins([0x1000] * 3, FT2)
w("ntuser_dirty_truncated.bin", a[:0x1000 + 0x2000 + 0x400])
# B: clean (88/88) SYSTEM, hbins 0x2000 + 0x1000, stored checksum corrupted
w("system_clean_badchecksum.bin",
  base(88, 88, FT, 5, 0, 0x3000, "SYSTEM", corrupt=0x00C0FFEE) + hbins([0x2000, 0x1000], FT))


# C: new-format .LOG1: 512-byte base block (type 6) + HvLE entries seq 670/671/672 with 2/1/3 dirty pages
def hvle(seq, npages, size):
    refs = b"".join(struct.pack("<II", i * 0x1000, 0x1000) for i in range(npages))
    hdr_len = (40 + len(refs) + 511) // 512 * 512
    total = hdr_len + npages * 0x1000
    h = b"HvLE" + struct.pack("<IIIIIQQ", total, 0, seq, size, npages, 0x1111111111111111 * seq % (1 << 64),
                              0x2222222222222222 * seq % (1 << 64)) + refs
    pages = b"".join(bytes([0x41 + i]) * 0x1000 for i in range(npages))
    return h.ljust(hdr_len, b"\0") + pages


c = base(673, 673, FT2, 6, 6, 0x3000, "\\??\\C:\\Users\\alice\\ntuser.dat", full=False)
c += hvle(670, 2, 0x3000) + hvle(671, 1, 0x3000) + hvle(672, 3, 0x3000) + b"\0" * 0x200
w("ntuser_log1_hvle.bin", c)
# D: old-format .LOG: base block (type 1) + DIRT vector (0x4000 / 4096 = 4 bytes, 7 bits set) + 7 sectors
d = base(41, 41, FT, 3, 1, 0x4000, "SOFTWARE", full=False)
vec = bytes([0b10110001, 0x03, 0x00, 0x80])
d += (b"DIRT" + vec).ljust(512, b"\0") + b"".join(bytes([0x61 + i]) * 512 for i in range(7))
w("software_log_dirt.bin", d)
# E: decoy -- a carved hbin with no base block (renamed NTUSER.DAT)
w("decoy_carved_hbin.bin", hbins([0x1000, 0x1000], FT))
