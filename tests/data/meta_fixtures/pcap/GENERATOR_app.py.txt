# Fixture writer for pcap_app: stdlib only (struct frames + ssl.MemoryBIO for genuine OpenSSL TLS 1.2 handshake
# bytes; the server cert is an `openssl req -x509` self-signed cert). Independent of lib/meta.
# usage: python3 GENERATOR_app.py out.pcap cert.pem key.pem
import ssl, struct, sys
out, certf, keyf = sys.argv[1:4]

def csum(b):
    if len(b) % 2: b += b"\0"
    s = sum(struct.unpack("!%dH" % (len(b)//2), b)); s = (s >> 16) + (s & 0xffff); s += s >> 16
    return (~s) & 0xffff

def ip4(src, dst, proto, payload):
    a = bytes(map(int, src.split("."))); b = bytes(map(int, dst.split(".")))
    h = struct.pack("!BBHHHBBH4s4s", 0x45, 0, 20+len(payload), 1, 0, 64, proto, 0, a, b)
    return h[:10] + struct.pack("!H", csum(h)) + h[12:] + payload

def tcp(sp, dp, seq, flags, data=b""):
    return struct.pack("!HHIIBBHHH", sp, dp, seq, 1, 0x50, flags, 65535, 0, 0) + data

def udp(sp, dp, data):
    return struct.pack("!HHHH", sp, dp, 8+len(data), 0) + data

def eth(l3):
    return b"\x00\x11\x22\x33\x44\x55" + b"\x66\x77\x88\x99\xaa\xbb" + b"\x08\x00" + l3

def qname(n):
    return b"".join(bytes([len(x)]) + x.encode() for x in n.split(".")) + b"\0"

def dnsq(tid, name, qt=1):
    return struct.pack("!HHHHHH", tid, 0x0100, 1, 0, 0, 0) + qname(name) + struct.pack("!HH", qt, 1)

def dnsr(tid, name, answers, rcode=0, qt=1):
    m = struct.pack("!HHHHHH", tid, 0x8180 | rcode, 1, len(answers), 0, 0) + qname(name) + struct.pack("!HH", qt, 1)
    for rt, rd in answers:   # owner name = pointer to the question (0xc00c)
        m += b"\xc0\x0c" + struct.pack("!HHIH", rt, 1, 300, len(rd)) + rd
    return m

# TLS 1.2 via OpenSSL MemoryBIO
cctx = ssl.SSLContext(ssl.PROTOCOL_TLS_CLIENT); cctx.check_hostname = False; cctx.verify_mode = ssl.CERT_NONE
cctx.set_alpn_protocols(["h2", "http/1.1"])
sctx = ssl.SSLContext(ssl.PROTOCOL_TLS_SERVER); sctx.maximum_version = ssl.TLSVersion.TLSv1_2
sctx.load_cert_chain(certf, keyf); sctx.set_alpn_protocols(["http/1.1"])
ci, co, si, so = ssl.MemoryBIO(), ssl.MemoryBIO(), ssl.MemoryBIO(), ssl.MemoryBIO()
cl = cctx.wrap_bio(ci, co, server_hostname="update.badcdn.example"); sv = sctx.wrap_bio(si, so, server_side=True)
try: cl.do_handshake()
except ssl.SSLWantReadError: pass
hello = co.read(); si.write(hello)
try: sv.do_handshake()
except ssl.SSLWantReadError: pass
flight = so.read()
