out = sys.argv[1]
def w(name, b):
    with open(os.path.join(out, name), "wb") as f: f.write(b)

         "ResultStatus": "Succeeded", "UserId": user, "Workload": wl}
    d.update(kw); return json.dumps(d, separators=(",", ":"))

# CloudTrail, S3 delivery (gzip {"Records": [...]})
def ct(t, src, name, region, ip, ut, user, **kw):
    r = {"eventVersion": "1.08", "userIdentity": {"type": ut, "principalId": "AIDAEXAMPLE", "arn":
         "arn:aws:iam::123456789012:user/" + user, "accountId": "123456789012", "userName": user},
         "eventTime": t, "eventSource": src, "eventName": name, "awsRegion": region, "sourceIPAddress": ip,
         "userAgent": "aws-cli/2.15.0", "recipientAccountId": "123456789012"}
    r.update(kw); return r

# Okta System Log API array
def ok(t, et, alt, ip, res, cc, reason=None):
    return {"uuid": "u-" + t, "published": t, "eventType": et, "version": "0", "severity": "INFO" if res == "SUCCESS" else "WARN",
            "displayMessage": et, "actor": {"id": "00u1", "type": "User", "alternateId": alt, "displayName": alt.split("@")[0]},
            "client": {"ipAddress": ip, "userAgent": {"rawUserAgent": "Mozilla/5.0"}, "geographicalContext": {"country": cc}},
            "outcome": {"result": res, "reason": reason}, "target": []}
